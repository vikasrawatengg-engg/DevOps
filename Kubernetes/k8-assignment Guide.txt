Create myapp-service.yaml

apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: myapp-ingress
  namespace: default
  annotations:
    nginx.ingress.kubernetes.io/rewrite-target: /
spec:
  ingressClassName: nginx
  rules:
  - host: <Public IP>.nip.io
    http:
      paths:
      - path: /
        pathType: Prefix
        backend:
          service:
            name: my-service   # ✅ Corrected name
            port:
              number: 80       # ✅ Matches your service port

kubectl apply -f myapp-ingress.yaml
 
kubectl get ingress
 
kubectl port-forward service/ingress-nginx-controller -n ingress-nginx --address 0.0.0.0 8080:80
